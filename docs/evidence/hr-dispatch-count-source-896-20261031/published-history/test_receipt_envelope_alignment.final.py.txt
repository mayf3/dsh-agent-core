RESULTS = []

def check(name, cond, detail=""):
    RESULTS.append((name, bool(cond), detail))


def envelope(mid="om_env_1", session="sess_real_1", target="agt_fake-agent",
             status="accepted", ok=True, extra=None):
    if not ok:
        return {"ok": False, "error": {"code": "outcome_unknown",
                                       "detail": "synthetic non-success"}}
    result = {"status": status, "targetAgentId": target,
              "sessionId": session, "messageId": mid}
    if extra:
        result.update(extra)
    return {"ok": True, "result": result}


class FakeSender:
    def __init__(self, receipt=None, behavior="ok"):
        self.calls = []
        self.receipt = receipt
        self.behavior = behavior

    def __call__(self, target, message, candidate):
        self.calls.append(target)
        if self.behavior == "raise":
            raise RuntimeError("synthetic transport failure")
        return self.receipt


def candidate(instance, target="agt_fake-agent", node="candidates",
              visit=None, projection="pX"):
    c = {"workflowInstanceId": instance, "currentNodeKey": node,
         "targetAgentId": target, "projection_updated_at": projection,
         "definitionKey": "fake-def-v1"}
    if visit:
        c["nodeVisitId"] = visit
    return c


def plan_for(*candidates):
    cands = list(candidates)
    return {"skips": [], "union": {"candidates": cands},
            "ranking": {"selected": cands, "deferred": []}}


def fresh_ledger():
    fd, path = tempfile.mkstemp(suffix=".jsonl")
    os.close(fd)
    os.unlink(path)
    return path


def last_record(path, instance, node="candidates", visit=None, projection="pX"):
    return fsm.get(path, instance, node, node_visit_id=visit,
                   projection_updated_at=projection)


def red_core_envelope_accepted():
    """Real nested envelope => ledger carries the reference from THIS return."""
    path = fresh_ledger()
    inst = "b3000000-0000-4000-8000-00000000000c"
    sender = FakeSender(receipt=envelope())
    out = v5.execute_round(plan_for(candidate(inst)),
                           sender, state_path=path, now_ms=T0)
    st = last_record(path, inst)
    check("E0a round item carried the raw envelope unchanged",
          out["dispatches"][0].get("receipt", {}).get("result", {}).get("messageId")
          == "om_env_1")
    rr = st.get("receiptRef") or {}
    check("E0b ledger persists reference from the nested envelope",
          rr.get("messageId") == "om_env_1" and rr.get("sessionId") == "sess_real_1",
          f"receiptRef={st.get('receiptRef')}")
    check("E0c context fields still come from the candidate, not the receipt",
          st.get("instanceId") == inst and st.get("agentId") == "agt_fake-agent"
          and st.get("nodeKey") == "candidates",
          f"row={ {k: st.get(k) for k in ('instanceId', 'agentId', 'nodeKey')} }")
    check("E0d reference marked REPORTED envelope transport result",
          rr.get("receiptSource") == "send_agent_return_envelope"
          and rr.get("reportedStatus") == "accepted"
          and "consumed" not in json.dumps(rr).lower()
          and "ack" not in json.dumps(rr).lower(),
          f"receiptRef={st.get('receiptRef')}")
    check("E0e report loss does not destroy the reference (ledger-only read)",
          rr.get("messageId") == "om_env_1",
          f"receiptRef={st.get('receiptRef')}")

def partial_or_invalid_coordinates():
    bad_results = [
        ("missing sessionId", {"ok": True, "result": {
            "status": "accepted", "targetAgentId": "agt_fake-agent",
            "messageId": "om_p1"}}),

        ("missing messageId", {"ok": True, "result": {
            "status": "accepted", "targetAgentId": "agt_fake-agent",
            "sessionId": "sess_x"}}),
        ("status not accepted", envelope(status="unknown")),
        ("result not an object", {"ok": True, "result": "accepted"}),
        ("ok missing", {"result": {"status": "accepted",

                                   "sessionId": "s", "messageId": "m"}}),

        path = fresh_ledger()

def new_shape_failure_closes_legacy_fallback():
    """R1: a recognized new-shape receipt failing validation is terminal
    UNKNOWN - stray top-level messageId/message_id/id keys must not
    re-admit it, and the ledger must show zero reference with the old
    count/fence semantics intact."""
    mixed_cases = [
        ("impostor ok+top-level id",
         {"ok": True, "messageId": "om_top", "status": "accepted"}),
        ("wrongtarget result + top-level id",
         {"ok": True, "result": {"status": "accepted",
                                 "targetAgentId": "agt_OTHER-agent",
                                 "sessionId": "s", "messageId": "m"},
          "id": "request-id"}),
        ("ok:false error + top-level messageId",
         {"ok": False, "error": {"code": "outcome_unknown"},
          "messageId": "om_e"}),
        ("partial result (missing sessionId) + top-level message_id",
         {"ok": True, "result": {"status": "accepted",
                                 "targetAgentId": "agt_fake-agent",
                                 "messageId": "om_p"},
          "message_id": "om_p"}),
    ]
    for i, (label, receipt) in enumerate(mixed_cases):
        path = fresh_ledger()
        inst = f"e6000000-0000-4000-8000-0000000000{i + 70:02x}"
        sender = FakeSender(receipt=receipt)
        v5.execute_round(plan_for(candidate(inst)), sender,
                         state_path=path, now_ms=T0 + i)
        st = last_record(path, inst)
        check(f"E5 {label}: closed to UNKNOWN, zero reference, count kept",
              st.get("dispatchCount") == 1 and not st.get("receiptRef"),
              f"count={st.get('dispatchCount')} receiptRef={st.get('receiptRef')}")

def opaque_coordinates_byte_exact():
    """R2: coordinates are validated and stored as original bytes - never
    stripped/normalized; whitespace-padded values are rejected by contract
    (never rewritten into a different ID); target binding is byte-exact."""
    # leading/trailing whitespace target must NOT match expected target
    for i, label, target in [(0, "trailing-space", "agt_fake-agent "),
                             (1, "leading-space", " agt_fake-agent")]:
        path = fresh_ledger()
        inst = f"f7000000-0000-4000-8000-00000000000{i + 1:x}"
        sender = FakeSender(receipt=envelope(
            mid=f"om_ws{i}", session=f"sess_ws{i}", target=target))
        v5.execute_round(plan_for(candidate(inst)), sender,
                         state_path=path, now_ms=T0 + i)
        st = last_record(path, inst)
        check(f"E10 {label} target: no byte-folding match, no reference",
              st.get("dispatchCount") == 1 and not st.get("receiptRef"),
              f"count={st.get('dispatchCount')} receiptRef={st.get('receiptRef')}")
    # whitespace-padded coordinates are explicitly rejected, not rewritten
    path = fresh_ledger()
    inst = "f7000000-0000-4000-8000-000000000003"
    sender = FakeSender(receipt=envelope(mid="  om_pad  ", session=" sess_pad "))
    v5.execute_round(plan_for(candidate(inst)), sender, state_path=path, now_ms=T0)
    st = last_record(path, inst)
    check("E10 padded coordinates: rejected by contract, not normalized",
          st.get("dispatchCount") == 1 and not st.get("receiptRef"),
          f"count={st.get('dispatchCount')} receiptRef={st.get('receiptRef')}")
    # accepted coordinates keep their ORIGINAL bytes (internal spaces kept)
    path = fresh_ledger()
    inst = "f7000000-0000-4000-8000-000000000004"
    sender = FakeSender(receipt=envelope(mid="om mid", session="sess 1"))
    v5.execute_round(plan_for(candidate(inst)), sender, state_path=path, now_ms=T0)
    rr = last_record(path, inst).get("receiptRef") or {}
    check("E11 accepted coordinates stored byte-exact (no rewrite)",
          rr.get("messageId") == "om mid" and rr.get("sessionId") == "sess 1"
          and rr.get("targetAgentId") == "agt_fake-agent",
          f"receiptRef={rr}")
    # missing/untrusted expected target: new binding stays UNKNOWN.  The
    # dispatcher cannot send without a target (the send itself would fail),
    # so this contract is probed at the adapter unit boundary, which is the
    # exact surface that decides promotion.
    no_target_cases = [
        ("absent expected target", None),
        ("empty expected target", ""),
        ("non-string expected target", 123),
    ]
    for i, (label, expected) in enumerate(no_target_cases):
        ref = v5._receipt_reference(envelope(), expected)
        check(f"E12 {label}: new binding stays UNKNOWN at adapter boundary",
              ref is None, f"ref={ref}")


def old_rows_never_backfilled():
    """R3: same-target/row positive is honest; explicit foreign-target and
    missing-binding envelope refs are rejected at the persistence entry;
    legacy-sourced refs keep historical semantics; historical rows are
    never backfilled."""
    path = fresh_ledger()
    legacy_inst = "a8000000-0000-4000-8000-000000000040"
    fsm.record_dispatch(path, legacy_inst, "candidates", dispatch_ts=T0,
                        agent_id="agt_legacy")
    inst = "a9000000-0000-4000-8000-000000000041"
    sender = FakeSender(receipt=envelope(mid="om_env_2", session="sess_real_2"))
    v5.execute_round(plan_for(candidate(inst)), sender, state_path=path,
                     now_ms=T0 + 1)
    own_ref = {"messageId": "om_env_2", "sessionId": "sess_real_2",
               "targetAgentId": "agt_legacy", "reportedStatus": "accepted",
               "receiptSource": "send_agent_return_envelope",
               "attemptBinding": "unknown"}
    fsm.record_dispatch(path, legacy_inst, "candidates", dispatch_ts=T0 + 2,
                        agent_id="agt_legacy", receipt_ref=own_ref)
    first = json.loads(open(path).read().splitlines()[0])
    check("E6a historical row stays without receiptRef (no backfill)",
          "receiptRef" not in first, f"first-row keys={sorted(first)}")
    last = last_record(path, legacy_inst, projection=None)
    rr = last.get("receiptRef") or {}
    check("E6b same-target envelope ref persists on its own row (own ref)",
          rr.get("messageId") == "om_env_2"
          and rr.get("targetAgentId") == "agt_legacy",
          f"receiptRef={rr}")
    st = last_record(path, inst)
    check("E6c envelope instance keeps its own reference (no cross-row leak)",
          (st.get("receiptRef") or {}).get("sessionId") == "sess_real_2",
          f"receiptRef={st.get('receiptRef')}")
    # foreign target on a legacy row: envelope-sourced ref must NOT persist
    path2 = fresh_ledger()
    foreign_ref = dict(own_ref, messageId="om_foreign",
                       targetAgentId="agt_fake-agent")
    fsm.record_dispatch(path2, "b1000000-0000-4000-8000-000000000042",
                        "candidates", dispatch_ts=T0, agent_id="agt_legacy",
                        receipt_ref=foreign_ref)
    st2 = last_record(path2, "b1000000-0000-4000-8000-000000000042",
                      projection=None)
    check("E6d foreign-target envelope ref rejected at persistence entry",
          not st2.get("receiptRef"),
          f"receiptRef={st2.get('receiptRef')}")
    # missing binding: envelope-source ref without explicit target / row agent
    path3 = fresh_ledger()
    unbound_ref = {k: v for k, v in own_ref.items() if k != "targetAgentId"}
    fsm.record_dispatch(path3, "b2000000-0000-4000-8000-000000000043",
                        "candidates", dispatch_ts=T0, agent_id="agt_legacy",
                        receipt_ref=unbound_ref)
    fsm.record_dispatch(path3, "b3000000-0000-4000-8000-000000000044",
                        "candidates", dispatch_ts=T0 + 1, agent_id=None,
                        receipt_ref=own_ref)
    st3a = last_record(path3, "b2000000-0000-4000-8000-000000000043",
                       projection=None)
    st3b = last_record(path3, "b3000000-0000-4000-8000-000000000044",
                       projection=None)
    check("E6e envelope ref without explicit target or without row agent: "
          "rejected (binding missing)",
          not st3a.get("receiptRef") and not st3b.get("receiptRef"),
          f"a={st3a.get('receiptRef')} b={st3b.get('receiptRef')}")
    # legacy-sourced ref without target contract: historical behavior kept
    path4 = fresh_ledger()
    legacy_ref = {"messageId": "om_old", "reportedStatus": "sent",
                  "receiptSource": "send_agent_return"}
    fsm.record_dispatch(path4, "b4000000-0000-4000-8000-000000000045",
                        "candidates", dispatch_ts=T0, agent_id=None,
                        receipt_ref=legacy_ref)
    st4 = last_record(path4, "b4000000-0000-4000-8000-000000000045",
                      projection=None)
    check("E6f legacy-sourced ref without target contract: unchanged",
          (st4.get("receiptRef") or {}).get("messageId") == "om_old",
          f"receiptRef={st4.get('receiptRef')}")

def receipts_never_cross_sessions():
    path = fresh_ledger()
    i1, i2 = "ac000000-0000-4000-8000-000000000060", "ad000000-0000-4000-8000-000000000061"
    v5.execute_round(plan_for(candidate(i1)),
                     FakeSender(receipt=envelope(mid="om_a", session="sess_a")),
                     state_path=path, now_ms=T0)
    v5.execute_round(plan_for(candidate(i2, target="agt_other-agent")),
                     FakeSender(receipt=envelope(mid="om_b", session="sess_b",
                                                 target="agt_other-agent")),
                     state_path=path, now_ms=T0 + 1)
    r1 = last_record(path, i1).get("receiptRef") or {}
    r2 = last_record(path, i2).get("receiptRef") or {}
    check("E9 two normal inputs each land in their own row "
          "(NOT a cross-attempt detection; attemptBinding stays 'unknown')",
          r1.get("messageId") == "om_a" and r1.get("sessionId") == "sess_a"
          and r2.get("messageId") == "om_b" and r2.get("sessionId") == "sess_b",
          f"r1={r1} r2={r2}")


def main():
    for fn in (red_core_envelope_accepted, envelope_field_discipline,
               target_binding_exact, partial_or_invalid_coordinates,
               non_success_envelopes_stay_unknown,
               new_shape_failure_closes_legacy_fallback,
               genuine_flat_legacy_contract_unchanged,
               opaque_coordinates_byte_exact,
               old_rows_never_backfilled,
               count_cooldown_and_failure_semantics_unchanged,
               receipts_never_cross_sessions):
        try:
            fn()
        except Exception as exc:
            RESULTS.append((fn.__name__ + " HARNESS", False, repr(exc)))
    failed = 0
    print(f"=== receipt envelope SOURCE-alignment probe "
          f"(module root: {ROOT}) ===")
    for name, ok, detail in RESULTS:
        print(f"{'PASS' if ok else 'FAIL'}  {name}  {detail}")
        failed += 0 if ok else 1
    print(f"=== {len(RESULTS) - failed}/{len(RESULTS)} passed, {failed} failed ===")
    sys.exit(1 if failed else 0)


if __name__ == "__main__":
    main()
