credential_pin_check() {
  local t i u g m mt
  t="$(stat -f '%HT' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  i="$(stat -f '%i' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  u="$(stat -f '%u' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  g="$(stat -f '%g' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  m="$(stat -f '%Lp' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  mt="$(stat -f '%m' "$CREDENTIAL_FILE" 2>/dev/null)" || return 1
  [ "$t" = "Regular File" ] && [ "$i" = "$PIN_CRED_INODE" ] \
    && [ "$u" = "$PIN_CRED_UID" ] && [ "$g" = "$PIN_CRED_GID" ] \
    && [ "$m" = "$PIN_CRED_MODE" ] && [ "$mt" = "$PIN_CRED_MTIME" ]
}

health_ok() {
  local body
  body="$(/usr/bin/curl -fsS --max-time 5 "$HEALTH_URL" 2>/dev/null)" || return 1
  printf '%s' "$body" | "$NODE_BIN" -e '

}

capture_log_sizes() {
  LOG_SIZE_PRE_RESTART="$(stat -f '%z' "$RUNTIME_LOG" 2>/dev/null || printf '0')"
  ERR_SIZE_PRE_RESTART="$(stat -f '%z' "$RUNTIME_ERR" 2>/dev/null || printf '0')"
}

appended_log_text() {
  {
    tail -c +$((LOG_SIZE_PRE_RESTART + 1)) "$RUNTIME_LOG" 2>/dev/null
    tail -c +$((ERR_SIZE_PRE_RESTART + 1)) "$RUNTIME_ERR" 2>/dev/null
  }
}

wait_runtime_healthy() {
  local max_s="$1" old_pid="$2" deadline pid
  deadline=$(( $(date +%s) + max_s ))
  while [ "$(date +%s)" -lt "$deadline" ]; do
    if health_ok; then
      pid="$(service_pid)"
      if [ -n "$pid" ] && [ "$pid" != "$old_pid" ]; then
        RUNTIME_PID_NEW="$pid"
        return 0
      fi
    fi
    sleep 2
  done
  return 1
}

function die(message) {

function digest(bytes) {

function objectJson(bytes, label) {
  let value;

function writeAll(fd, bytes) {
  let offset = 0;
  while (offset < bytes.length) {
    const n = fs.writeSync(fd, bytes, offset, bytes.length - offset, null);
    if (!Number.isInteger(n) || n <= 0) die('short-write');

}
function syncDirectory(dir) {

rollback() {
  local reason="$1" old_pid rollback_pid appended current_pid out stale
  local bad=""
  local rb_map_bytes=0 rb_map_sha=0 rb_key_absent=0 rb_map_metadata=0 rb_map_count=0
  local rb_home_mode=0 rb_credential=0 rb_restart=0 rb_health=0 rb_pid_stable=0
  local rb_boot_absent=0 rb_fallback=0 rb_marker_absent=0 rb_stale_absent=1
  [ "$ROLLBACK_ACTIVE" = "1" ] && return 4
  ROLLBACK_ACTIVE=1
  trap - EXIT
  log "── ROLLBACK (reason: $reason)"

  if /bin/rm -f "$MARKER_FILE" 2>/dev/null && [ ! -e "$MARKER_FILE" ]; then rb_marker_absent=1; else bad="${bad}\n- canary marker still present"; fi

  if [ "$MAP_MUTATION_POSSIBLE" = "1" ]; then
    if [ -n "$BACKUP_FILE" ] && [ -f "$BACKUP_FILE" ]; then
      if out="$(run_atomic_as_authsvc restore "$MAP_FILE" "$BACKUP_FILE" "$AGENT_ID" "$TARGET_WS" "$PIN_MAP_ENTRIES" "$PRE_SHA256" "-" "-" 2>&1)"; then :; else bad="${bad}\n- mapping restore transaction failed: $out"; fi
    else
      bad="${bad}\n- mapping restore prerequisites unavailable"
    fi
  fi

  [ -n "$BACKUP_FILE" ] && [ -f "$BACKUP_FILE" ] && /usr/bin/cmp -s "$BACKUP_FILE" "$MAP_FILE" && rb_map_bytes=1 || bad="${bad}\n- mapping exact pre-bytes mismatch"
  [ "$(sha256_file "$MAP_FILE" 2>/dev/null)" = "$PRE_SHA256" ] && rb_map_sha=1 || bad="${bad}\n- mapping SHA mismatch"
  map_key_absent_check && rb_key_absent=1 || bad="${bad}\n- CTO mapping key is not absent"
  map_count_check "$PIN_MAP_ENTRIES" && rb_map_count=1 || bad="${bad}\n- mapping entry count mismatch"
  if [ "$(map_uid)" = "$PIN_MAP_UID" ] && [ "$(map_gid)" = "$PIN_MAP_GID" ] && [ "$(map_mode)" = "$PIN_MAP_MODE" ]; then rb_map_metadata=1; else bad="${bad}\n- mapping owner/group/mode mismatch"; fi

  if [ "$HOME_MODE_CHANGED" = "1" ]; then change_home_mode 755 2>/dev/null || bad="${bad}\n- home mode restore command failed"; fi
  [ "$(stat -f '%Lp' "$HOME_DIR" 2>/dev/null)" = "755" ] && rb_home_mode=1 || bad="${bad}\n- home numeric mode is not 755"
  credential_pin_check && rb_credential=1 || bad="${bad}\n- credential metadata pin mismatch"

  old_pid="$(service_pid)"
  capture_log_sizes
  if restart_service 2>/dev/null; then
    rb_restart=1
    if wait_runtime_healthy 180 "$old_pid"; then
      rb_health=1
      rollback_pid="$RUNTIME_PID_NEW"
      stability_pause
      current_pid="$(service_pid)"
      if health_ok; then :; else rb_health=0; bad="${bad}\n- rollback HEALTH_URL did not remain healthy"; fi
      if [ "$rb_health" = "1" ] && [ -n "$rollback_pid" ] && [ "$current_pid" = "$rollback_pid" ] && [ "$rollback_pid" != "$old_pid" ]; then rb_pid_stable=1; else bad="${bad}\n- rollback runtime PID is not updated and stable"; fi
    else
      bad="${bad}\n- rollback HEALTH_URL/new-pid check failed"
    fi
  else
    bad="${bad}\n- rollback service restart command failed"
  fi

  if [ "$rb_health" = "1" ]; then
    appended="$(appended_log_text)"
    if ! printf '%s' "$appended" | /usr/bin/grep -qF 'primary workspace imports loaded'; then bad="${bad}\n- rollback boot evidence is missing import composition proof"
    elif printf '%s' "$appended" | /usr/bin/grep -qF "$AGENT_ID -> $TARGET_WS"; then bad="${bad}\n- rollback boot evidence still contains target mapping"
    else rb_boot_absent=1; fi
    if fallback_resolution_check; then rb_fallback=1; else bad="${bad}\n- deployed resolver does not return exact fallback path"; fi
  fi

  for stale in "$CHILD_PID" "$RUNTIME_PID_OLD" "$old_pid"; do
    if [ -n "$stale" ] && [ "$stale" != "${rollback_pid:-}" ] && /bin/kill -0 "$stale" 2>/dev/null; then rb_stale_absent=0; bad="${bad}\n- stale pre-rollback process remains alive: $stale"; fi
  done

  log "ROLLBACK_CHECK MAP_EXACT_PRE_BYTES=$rb_map_bytes"
  log "ROLLBACK_CHECK MAP_SHA256_PRE=$rb_map_sha"
  log "ROLLBACK_CHECK CTO_KEY_ABSENT=$rb_key_absent"
  log "ROLLBACK_CHECK MAP_OWNER_GROUP_MODE=$rb_map_metadata"
  log "ROLLBACK_CHECK MAP_ENTRY_COUNT=$rb_map_count"
  log "ROLLBACK_CHECK HOME_MODE_755=$rb_home_mode"
  log "ROLLBACK_CHECK CREDENTIAL_PIN=$rb_credential"
  log "ROLLBACK_CHECK RESTART_SUCCEEDED=$rb_restart"
  log "ROLLBACK_CHECK HEALTH_URL_OK=$rb_health"
  log "ROLLBACK_CHECK RUNTIME_PID_UPDATED_STABLE=$rb_pid_stable"
  log "ROLLBACK_CHECK BOOT_MAPPING_ABSENT=$rb_boot_absent"
  log "ROLLBACK_CHECK FALLBACK_PROOF=$rb_fallback"
  log "ROLLBACK_CHECK MARKER_ABSENT=$rb_marker_absent"
  log "ROLLBACK_CHECK STALE_PROCESSES_ABSENT=$rb_stale_absent"

  if [ -z "$bad" ] && [ "$rb_map_bytes" = "1" ] && [ "$rb_map_sha" = "1" ] && [ "$rb_key_absent" = "1" ] \
    && [ "$rb_map_metadata" = "1" ] && [ "$rb_map_count" = "1" ] && [ "$rb_home_mode" = "1" ] \
    && [ "$rb_credential" = "1" ] && [ "$rb_restart" = "1" ] && [ "$rb_health" = "1" ] \
    && [ "$rb_pid_stable" = "1" ] && [ "$rb_boot_absent" = "1" ] && [ "$rb_fallback" = "1" ] \
    && [ "$rb_marker_absent" = "1" ] && [ "$rb_stale_absent" = "1" ]; then
    log "RESULT: FAILED_AND_ROLLED_BACK — $reason"
    log "FINAL: FAILED_AND_ROLLED_BACK; EXIT=1"
    exit 1
  fi

  log "RESULT: ROLLBACK_INCOMPLETE — $reason"
  printf '%b\n' "UNRESTORED_STATE:${bad}"
  log "FINAL: ROLLBACK_INCOMPLETE; EXIT=4; BACKUP=${BACKUP_DIR:-unavailable}"
  exit 4
}

fail() {
  if [ "$WRITES_STARTED" = "1" ]; then rollback "$1"; else zero_write_exit "$1"; fi
}

  log "FINAL: SUCCESS; EXIT=0; SYNTHETIC_CASE=$scenario; ROLLBACK_CALLS=0"
  exit 0
