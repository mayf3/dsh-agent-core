import pathlib,json,hashlib,datetime,subprocess,re,os,sys
ROOT=pathlib.Path('/private/tmp/dsh-523-integration')
EV=ROOT/'docs/evidence/guard-integration-523-r3-20261104'
def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def snapshot(label):
    manifests=[]
    nm=ROOT/'node_modules'
    for p in sorted(nm.glob('*/package.json'))+sorted(nm.glob('@*/*/package.json')):
        data=json.loads(p.read_text());manifests.append(dict(name=data.get('name'),version=data.get('version'),path=str(p),realpath=str(p.resolve()),sha256=sha(p)))
    links=[]
    for start in [nm/'@agent-core',pathlib.Path('/tmp/home/agt_a/profiles/node_modules'),pathlib.Path('/tmp/home/agt_delivery-v0-1/profiles/node_modules')]:
        for parent,dirs,files in os.walk(start,followlinks=False):
            for name in sorted(dirs+files):
                p=pathlib.Path(parent)/name
                if p.is_symlink(): links.append(dict(path=str(p),target=os.readlink(p),realpath=str(p.resolve()),exists=p.exists()))
    result=dict(at=now(),sourceHead=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),nodeModulesIsSymlink=nm.is_symlink(),nodeModulesRealpath=str(nm.resolve()),packages=manifests,links=links)
    (EV/('dependency-and-links.'+label+'.json')).write_text(json.dumps(result,indent=2)+'\n')
    return result
def run(label,cmd):
    prev=json.loads((ROOT/'docs/evidence/guard-integration-523-r2-20261104/baseline-check.json').read_text())
    env={k:prev['env'][k] for k in ('PATH','TMPDIR')}
    runtime=EV/('runtime-'+label);runtime.mkdir();env['PRODUCTION_RUNTIME_ROOT']=str(runtime)
    log=EV/(label+'.tap');rec=dict(command=cmd,env=env,cwd=str(ROOT),sourceHead=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),startedAt=now(),log=log.name,node=subprocess.check_output(['node','--version'],text=True).strip())
    with log.open('wb') as f: result=subprocess.run(cmd,cwd=ROOT,env=env,stdout=f,stderr=subprocess.STDOUT)
    rec.update(exitCode=result.returncode,finishedAt=now(),bytes=log.stat().st_size,sha256=sha(log))
    (EV/(label+'.json')).write_text(json.dumps(rec,indent=2)+'\n')
    print(json.dumps(rec),flush=True); print('\n'.join(log.read_text().splitlines()[-10:]),flush=True)
    return result.returncode
if __name__=='__main__':
    action=sys.argv[1]
    if action=='snapshot': snapshot(sys.argv[2])
    elif action=='focused':
        assert json.loads((EV/'install.json').read_text())['exitCode']==0
        snapshot('pre-focused')
        prior=json.loads((ROOT/'docs/evidence/guard-integration-523-r2-20261104/comparison.json').read_text())
        old=json.loads((ROOT/'docs/evidence/guard-integration-523-r2-20261104/integration-results.json').read_text())
        titles=prior['integrationOnlyFailures']; mapping={r['title']:r for r in old['results']}
        # Preserve original test-file/TAP order; one exact test per process.
        for i,title in enumerate(sorted(titles,key=lambda t:mapping[t]['tapId']),1):
            loc=mapping[title]['location']; filename=loc.rsplit(':',2)[0]
            run('focused-'+str(i),['node','--test','--test-reporter=tap','--test-name-pattern=^'+re.escape(title)+'$',filename])
        snapshot('post-focused')
    elif action=='guard':
        raise SystemExit(run('guard',['node','--test','--test-reporter=tap','packages/agent-router/test/route-chain/switch-workspace-guard.test.js']))
    elif action=='package':
        snapshot('pre-package')
        rc=run('agent-router',['node','--test','--test-reporter=tap','packages/agent-router/test/*.test.js','packages/agent-router/test/*/*.test.js'])
        snapshot('post-package')
        raise SystemExit(rc)
